cmake_minimum_required(VERSION 3.16)
project(web-security-inspector LANGUAGES CXX)
if(NOT WIN32)
    message(FATAL_ERROR "This application requires Windows.")
endif()
add_executable(web-security-inspector WIN32 main.cpp inspector.cpp)
target_compile_features(web-security-inspector PRIVATE cxx_std_17)
target_link_libraries(web-security-inspector PRIVATE winhttp crypt32 shlwapi comdlg32)
if(MSVC)
    target_compile_options(web-security-inspector PRIVATE /W4 /utf-8 /EHsc /permissive-)
else()
    target_compile_options(web-security-inspector PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(MINGW)
    target_link_options(web-security-inspector PRIVATE -municode)
endif()
